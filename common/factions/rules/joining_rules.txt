		}
	}
	available = {
		IF = {
			limit = {

			}

					is_major = yes
					has_government = fascism

		IF = {
			limit = {
				ROOT = { original_tag = JAP }
			}
			always = no 
		}
		ELSE = {
			always = yes 
		}
		
	}
	trigger = {
		date>1960.1.1
	}
}
