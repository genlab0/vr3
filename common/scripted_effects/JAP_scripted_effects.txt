JAP_convert_civs = {
    random_owned_controlled_state = {
		limit = { industrial_complex > 1 }
		remove_building = {
			type = industrial_complex
			level = 1
		}
		add_building_construction = {
			type = arms_factory
			level = 1
			instant_build = yes
		}
	}
}

JAP_increase_inflation = {

	if = {
		limit = { has_idea = JAP_inflation_1 }
		swap_ideas = {
			remove_idea = JAP_inflation_1
			add_idea = JAP_inflation_2
		}
