FRA_initiate_legitimacy_power_balance = {
	set_power_balance = {
		id = FRA_legitimacy_power_balance
		left_side = FRA_low_legitimacy # will keep the last set side if not specified
		right_side = FRA_high_legitimacy # will keep the last set side if not specified
		set_value = -0.8 # if specified will set the value of the power balance, this option will override the effect of set_default so the two options should not be used together. if neither set_default or set_value is specified the current value of the power balance will not change
	}
}

FRA_bop_medium_increase_effect = {

		value = 0.15
	}

remove_worker_shortage_effect = {	
	if = {
		limit = { has_idea = FRA_worker_shortage }
		remove_ideas = FRA_worker_shortage
	}
	else_if = {
		limit = { has_idea = FRA_full_employment }
		remove_ideas = FRA_full_employment
	}
}
